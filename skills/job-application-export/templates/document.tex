% Pandoc template for Bewerbungsdokumente (Lebenslauf, Anschreiben).
% Used by scripts/export_document.py with --template. Requires LuaLaTeX (decision O-3).
%
% Design (spec/job-application/lebenslauf/en.md LB-11): one column, one accent colour, thin rules under
% section headings, no tables, no icons, no text in header or footer except a quiet page number. Colour and
% rules do not change the text a parser reads (spec R-8.3). Font: Lato when installed (package fonts-lato),
% Latin Modern Sans otherwise.
%
% Sizes: name 25 pt, section heading 13 pt, position 11 pt, body 10.5 pt on 14.3 pt leading, period and
% employer 9.5 pt. The macros \cvheader, \cvstation and \cvrow are emitted by templates/cv.lua.
\documentclass[11pt,a4paper]{article}
\usepackage[ngerman]{babel}
\usepackage{fontspec}
\usepackage{xcolor}
$if(letter)$
\usepackage[left=25mm,right=20mm,top=20mm,bottom=20mm]{geometry}
$else$
\usepackage[left=20mm,right=18mm,top=17mm,bottom=19mm,footskip=9mm]{geometry}
$endif$
\usepackage{hyperref}
% longtable and booktabs are loaded only because pandoc's LaTeX writer may reference them; the Lebenslauf has
% no tables (spec R-8.3, CV-08), so a table in the source is a finding, not a layout element.
\usepackage{longtable,booktabs}
\usepackage{graphicx}
\usepackage{microtype}

% Colours: ink 15.8:1, accent 8.7:1 and muted 6.1:1 against white; all three stay readable in grayscale.
\definecolor{accent}{HTML}{1F4E79}
\definecolor{ink}{HTML}{1E2328}
\definecolor{muted}{HTML}{5A6270}

\IfFontExistsTF{Lato}{\setmainfont{Lato}}{\setmainfont{Latin Modern Sans}}
\color{ink}

\hypersetup{colorlinks=true,urlcolor=accent,linkcolor=accent}

\makeatletter
\renewcommand{\normalsize}{\@setfontsize\normalsize{10.5}{14.3}}
\renewcommand{\small}{\@setfontsize\small{9.5}{12.6}}
\renewcommand{\footnotesize}{\@setfontsize\footnotesize{9}{11.5}}
\renewcommand{\large}{\@setfontsize\large{13}{16}}
\normalsize

% Ragged right with a bounded stretch: unlike \raggedright (infinite stretch) it lets TeX balance the lines,
% so a paragraph rarely ends in a single word. Hyphenation stays rare. \@rightskip makes lists inherit it.
\def\cvragged{\let\\\@centercr\@rightskip=0pt plus 5em\relax\rightskip\@rightskip\leftskip\z@skip\parindent\z@}
\cvragged
\tolerance=1500
\emergencystretch=2em
\hyphenpenalty=1500
\exhyphenpenalty=300
\lefthyphenmin=3
\righthyphenmin=3
\widowpenalty=10000
\clubpenalty=10000
\raggedbottom

\setlength{\parindent}{0pt}
\setlength{\parskip}{0.45em}
\setcounter{secnumdepth}{0}
\providecommand{\tightlist}{\setlength{\itemsep}{2pt}\setlength{\parskip}{0pt}}

% Page number: quiet, centred, no other footer content.
\def\ps@plain{\let\@oddhead\@empty\def\@oddfoot{\hfil{\footnotesize\color{muted}\thepage}\hfil}\let\@evenhead\@empty\let\@evenfoot\@oddfoot}
\pagestyle{plain}

% Bullets: accent-coloured, narrower indent, tight spacing inside a list.
\setlength{\leftmargini}{1.2em}
\setlength{\leftmarginii}{1.1em}
\renewcommand{\labelitemi}{\textcolor{accent}{\textbullet}}
\renewcommand{\labelitemii}{\textcolor{muted}{\textendash}}
\setlength{\topsep}{0pt}
\setlength{\partopsep}{0pt}
\setlength{\itemsep}{2pt}

% Grid: a left column for periods and labels, the content to its right.
\newlength{\cvperiodwidth}
\setlength{\cvperiodwidth}{3.3cm}

% Start a new page when less than #1 is left, so a heading is never the last thing on a page (LB-11.3, values
% LB-11.11). Directly after a heading (@nobreak set by \@afterheading) no break is offered at all: the heading
% has reserved the space, and a break here would leave it alone at the foot of the page.
\newcommand{\cvneedspace}[1]{\par\if@nobreak\else\penalty-100\begingroup\setlength{\dimen@}{#1}\dimen@ii\pagegoal\advance\dimen@ii-\pagetotal\ifdim\dimen@>\dimen@ii\break\fi\endgroup\fi}

% Headings end with \@afterheading, as LaTeX's own sectioning does: it sets @nobreak, so the list or paragraph
% that follows adds no \@beginparpenalty (a legal, even encouraged break point) between heading and body.
% The first paragraph or \item after the heading clears the flag again.
% Section heading: text and rule form one box.
\newcommand{\cvsection}[1]{\vbox{\offinterlineskip\hbox{\large\bfseries\color{accent}#1}\kern3pt\hbox{\color{accent}\rule{\linewidth}{0.7pt}}}}
\renewcommand{\section}[1]{\par\addvspace{0.6em}\hbox{\large\bfseries\color{accent}#1}\par\nobreak\vspace{0.6em}\@afterheading}
\renewcommand{\subsection}[1]{\par\cvneedspace{6\baselineskip}\addvspace{1.2em}\cvsection{#1}\par\nobreak\vspace{0.2em}\@afterheading}
\renewcommand{\subsubsection}[1]{\par\cvneedspace{3.5\baselineskip}\addvspace{0.8em}{\bfseries #1}\par\nobreak\vspace{0.1em}\@afterheading}

% Header block: title label, name, then the personal data in the grid.
\newcommand{\cvheader}[4]{%
  \par\noindent{\small\bfseries\color{accent}#1}\par\vspace{2pt}%
  \noindent{\fontsize{25}{29}\selectfont\bfseries\color{ink}#2}\par\vspace{5pt}%
  \noindent\makebox[\cvperiodwidth][l]{\small\color{muted}#3}\parbox[t]{\dimexpr\linewidth-\cvperiodwidth\relax}{\cvragged\small #4}\par}

% Station: period in the left column, position bold, employer below in the muted colour. An empty third
% argument (a gap entry such as "Zeitraum | Elternzeit") leaves out the employer line instead of an empty one.
% The head reserves 3.5 lines (LB-11.11) and, like a heading, allows no break before the bullets that follow.
\newcommand{\cvstation}[3]{%
  \par\cvneedspace{3.5\baselineskip}\addvspace{0.8em}\begingroup\parskip=0pt
  \hyphenpenalty=10000\exhyphenpenalty=10000
  \noindent\makebox[\cvperiodwidth][l]{\small\color{muted}#1}%
  \parbox[t]{\dimexpr\linewidth-\cvperiodwidth\relax}{\cvragged\fontsize{11}{14.3}\selectfont\bfseries #2%
    \if\relax\detokenize{#3}\relax\else\\[-1pt]{\small\normalfont\color{muted}#3}\fi}\par
  \endgroup\nobreak\vspace{0.15em}\@afterheading}

% Row: a period in the left column and its text to the right (education, certificates).
\newcommand{\cvrow}[2]{%
  \par\noindent\makebox[\cvperiodwidth][l]{\small\color{muted}#1}%
  \parbox[t]{\dimexpr\linewidth-\cvperiodwidth\relax}{\cvragged #2}\par\vspace{2pt}}
\makeatother

$if(title)$\title{$title$}$endif$
$if(author)$\author{$author$}$endif$
\date{}
$if(keywords)$\hypersetup{pdfkeywords={$keywords$}}$endif$
$if(trailerid)$\directlua{pdf.settrailerid("[<$trailerid$> <$trailerid$>]")}$endif$
\begin{document}
$if(title)$\maketitle$endif$
$body$
\end{document}
